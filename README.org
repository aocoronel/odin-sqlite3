* SQLite3 bindings for Odin

#+BEGIN_QUOTE
This project was hard forked from [[https://github.com/flysand7/odin-sqlite3][odin-sqlite3]] by [[https://github.com/flysand7][Sunagatov Denis]], who did port most part of the bindings.

These bindings were hand-written and thus can contain many mistakes. In case you found a mistake consider [[https://github.com/aocoronel/odin-sqlite3/issues][opening up an issue]] or making a [[https://github.com/aocoronel/odin-sqlite3/pulls][pull request]].

*Progress:* missing to port SQLite Sessions and FTS5. If you don't need any of these, you can already use it.
#+END_QUOTE

#+HTML: <details>
#+HTML:   <summary>Project structure</summary>
- ~amalgamation/~: SQLite3's official amalgamation of the source code. Replace the contents of the folder with fresh version of amalgamation to update the version.
- ~sqlite3/~: Raw bindings of SQLite3, and addons.
- ~test/~: Test addons.
#+HTML: </details>

#+HTML: <details>
#+HTML:   <summary>Convention</summary>
- Declarations happens in the same order as ~sqlite.h~.
- Declarations persist the original C header comments.
- Removed all sqlite/SQLITE prefixes.
- No legacy procedures.
- Naming conventions follows Odin's.
- Type names follow the original, or if doesn't exist they are adapted.
- Procedure names follow the original.
#+HTML: </details>

** Using

Clone the repository.

#+BEGIN_SRC bash
git clone https://github.com/aocoronel/odin-sqlite3
#+END_SRC

Compile SQLite3 from source to ~sqlite3/~, or install SQLite3 in the system.

- Windows:
  - MSVC: ~./build.cl.bat~
  - Clang: ~./build-clang-windows~
- Unix:
  - GCC: ~./build-gcc.sh~
  - Clang: ~./build-clang.sh~

Move the ~sqlite3~ directory to your project, or to the Odin ~shared/~ library, which you will find in ~odin root~.

Importing the library:

#+BEGIN_SRC odin
  import "shared:sqlite3"

  // If installed in shared/
  odin run .

  // Otherwise
  odin run . -collection:shared=path/to/shared
  // NOTE: 'shared' is an optional name, but just happens to be convenient
  // if you want both to have sqlite3 locally and globally
#+END_SRC

** Example

#+BEGIN_QUOTE
This uses the API provided in [[.sqlite3/wrapper.odin]] and [[./sqlite3/exec.odin]] which isn't part of SQLite3. Otherwise, use the official [[https://sqlite.org/docs.html][SQLite3 documentation]], for more details. Alternatively, the [[./sqlite3/sqlite3.odin]] contains all the original C header comments.
#+END_QUOTE

All addons provided accepts the schema type to be ~cstring~, ~string~, ~cstring16~ and ~string16~.

*** Open the database:

#+BEGIN_SRC odin
DB_FILE :: "test/db.sqlite"
db: ^sqlite3.SQLite3
defer sqlite3.close(db) // should always be called
rc := sqlite3.open(DB_FILE, &db)
if rc != nil {
	log.errorf("Unable to open database '%s'(%v): %s", DB_FILE, rc, sqlite.errmsg(db))
	return
}
// Do things with database
#+END_SRC

*** Run a simple query to create some tables:

#+BEGIN_SRC odin
rc = sqlite.exec(db, `
    CREATE TABLE IF NOT EXISTS users (
        id INTEGER PRIMARY KEY,
        name VARCHAR(64) NOT NULL,
        flag INTEGER NOT NULL
   );`
)
#+END_SRC

Insert some users.

#+BEGIN_SRC odin
rc = sqlite.exec(db, `
    INSERT INTO users (name, flag) VALUES
        (?, ?),
        (?, ?),
        (?, ?),
        (?, ?);`,
    "john", 1,
    "mary", 0,
    "alice", 1,
    "bob", 0)
)
#+END_SRC

Which is equivalent to the following:

#+BEGIN_SRC odin
  stmt, rc := sqlite.prepare(db, `
      INSERT INTO users (name, flag) VALUES
          (?, ?),
          (?, ?),
          (?, ?),
          (?, ?);`,
      "john", 1,
      "mary", 0,
      "alice", 1,
      "bob", 0)
  sqlite.exec(stmt)

  // When the user creates que stmt manually, they also have to finalize it
  rc = sqlite.finalize(stmt)
#+END_SRC

Reusing the same statement:

#+BEGIN_SRC odin
  stmt, rc := sqlite.prepare(db, `
      INSERT INTO users (name, flag) VALUES
          (?, ?);`,
      "mary", 0)
  sqlite.exec(stmt)

  sqlite.reset(stmt); sqlite.clear_bindings(stmt);
  bind(stmt, "alice", 1)

  sqlite.exec(stmt)

  rc = sqlite.finalize(stmt)
#+END_SRC

*** Iterate select results:

#+BEGIN_SRC odin
  // Create prepared statement
  stmt, rc := sqlite.prepare(db, `
      INSERT into users (name, flag) VALUES
         (?, ?),
         (?, ?)
      RETURNING *`, "douglas", 1, "jonathan", 0)

  // Iterate the results
  for {
	rc = step(stmt)
	if rc != .Row do break
	row := sqlite.exec_dynamic(stmt)
	defer delete(row)
	fmt.println(row)
  }
  sqlite.finalize(stmt)
#+END_SRC

Alternatively, the iteration can be performed with structs.

#+BEGIN_SRC odin
  // Create prepared statement
  stmt, rc := sqlite.prepare(db, `
      INSERT into users (name, flag) VALUES
         (?, ?),
         (?, ?)
      RETURNING *`, "douglas", 1, "jonathan", 0)

  // Iterate the results
  for {
	rc = step(stmt)
	if rc != .Row do break
	row: struct{ name: string, flag: bool }
	row, rc = sqlite.exec_struct(db, stmt, struct { name: string, flag: bool })
	fmt.println(row)
  }
  sqlite.finalize(stmt)
#+END_SRC

*** Collect all results

#+BEGIN_SRC odin
  rows, rc := sqlite.exec_dynamic(db, `
      INSERT into users (name, flag) VALUES
         (?, ?),
         (?, ?)
      RETURNING *`, "douglas", 1, "jonathan", 0)
  defer {
  	for row in rows {
  		delete(row)
  	}
  	delete(rows)
  }
#+END_SRC

Alternatively with structs:

#+BEGIN_SRC odin
  rows, status := sqlite.exec_struct(db,
      struct { name: string, flag: bool }, `
      INSERT into users (name, flag) VALUES
         (?, ?),
         (?, ?)
      RETURNING *`, "douglas", 1, "jonathan", 0)
  defer delete(rows)
#+END_SRC

*** Getting the columns and rows

#+BEGIN_SRC odin
  dyn_cols: [dynamic]cstring
  cols, vals, rc := sqlite.exec_column(db,
      &dyn_cols,
      `INSERT into users (name, flag) VALUES
         (?, ?),
         (?, ?)
      RETURNING *`, "douglas", 1, "jonathan", 0)
  defer {
  	for col in cols {
  		delete(col)
  	}
  	delete(cols)
  }
#+END_SRC

*** Iterating statements

All ~exec_*~ procedures above execute only one statement at a time. If you have a schema like the following, there is an extra step to execute everything.

#+BEGIN_SRC odin
  SCHEMA ::
  ```
  INSERT INTO users (name, flag) VALUES ('bob', 1); -- 1
  SELECT * from users; -- 2
  ```
#+END_SRC

You can either run everything at once, without arguments as follow:

#+BEGIN_SRC odin
  rc := sqlite.exec_all(db, SCHEMA)
  rc := sqlite.exec_all(db, SCHEMA, callback, nil)
#+END_SRC

Or you can use any of the previous ~exec_*~ procedures, but pass a implicit parameter called ~tail~.

#+BEGIN_SRC odin
  /* using the tail without exec_* */
  tail: string = SCHEMA
  stmt, rc := sqlite.prepare(db, tail, tail = &tail)
  sqlite.exec(stmt)
  sqlite.finalize(stmt) // always finalize

  if !sqlite.is_sql_valid(tail) do return
  stmt, rc = sqlite.prepare(db, tail, tail = &tail)
  sqlite.exec(stmt)
  sqlite.finalize(stmt)

  /* using the tail with exec_* */
  tail: cstring = SCHEMA
  rc := sqlite.exec(db, tail, tail = &tail)

  if !sqlite.is_sql_valid(tail) do return
  rc := sqlite.exec(db, tail, tail = &tail)

  /* but likely you are going to do this */
  tail: string = SCHEMA
  for rc == .Ok && sqlite.is_sql_valid(tail) {
  	rc = sqlite.exec(db, tail, tail = &tail)
  }
#+END_SRC
